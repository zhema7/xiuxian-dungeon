export type SpawnEntry = [string, number, number]

/** 楼层刷怪表：1~6 层读配置；7 层起为无尽妖域动态生成（每 3 层一位 Boss，轮换出场） */
export function spawnTableFor(floor: number): SpawnEntry[] {
  const fixed = FLOOR_SPAWNS[floor]
  if (fixed) return fixed
  const depth = floor - 6
  if (depth % 3 === 0) {
    const bossId = ['boss', 'boss2', 'boss3'][Math.floor(depth / 3) % 3]
    return [[bossId, 1, 1], ['eagle', 1, 2], ['wraith', 1, 2], ['moth', 0, 1]]
  }
  const extra = Math.min(3, Math.floor(depth / 2))
  const table: SpawnEntry[] = [
    ['slime', 1, 2], ['bat', 1, 2 + extra], ['spitter', 1, 2],
    ['wraith', 1 + extra, 2 + extra], ['flower', 0, 1 + Math.min(2, extra)], ['tortoise', 1, 1 + extra],
  ]
  // 深层加入妖域新怪
  if (depth >= 3) table.push(['eagle', 1, 1 + extra])
  if (depth >= 5) table.push(['moth', 1, 1 + Math.min(2, extra)])
  if (depth >= 7) table.push(['golem', 1, 1])
  return table
}

// mulberry32 种子随机
export function makeRng(seed: number) {
  let a = seed >>> 0
  return function () {
    a |= 0; a = (a + 0x6d2b79f5) | 0
    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

export function randInt(rng: () => number, min: number, max: number) {
  return min + Math.floor(rng() * (max - min + 1))
}

export function pickWeighted<T>(rng: () => number, entries: Array<[T, number]>): T {
  const total = entries.reduce((s, e) => s + e[1], 0)
  let r = rng() * total
  for (const [v, w] of entries) { r -= w; if (r <= 0) return v }
  return entries[entries.length - 1][0]
}

export interface DamageResult { dmg: number; crit: boolean }

export function calcDamage(
  atk: number, mult: number, critRate: number,
  def: number, rng: () => number,
): DamageResult {
  const crit = rng() < critRate
  let raw = atk * mult * (crit ? 1.6 : 1)
  raw *= 0.9 + rng() * 0.2 // ±10% 浮动
  const dmg = Math.max(1, Math.round(raw - def))
  return { dmg, crit }
}

export interface EquipItem {

export const ENH_MAX = 5
/** 强化费用（灵石） */
export function enhanceCost(it: EquipItem): number {
  return 25 * ((it.enh ?? 0) + 1) + it.quality * 15
}
/** 强化后基座数值倍率（每级 +10%） */
export function enhMult(it: EquipItem): number {
  return 1 + 0.1 * (it.enh ?? 0)
}

export function normalizeItem(it: EquipItem): EquipItem {
  it.maxHp = it.maxHp ?? 0
  it.affixes = it.affixes ?? []
  it.enh = it.enh ?? 0
  return it
}

function roundAffixValue(stat: ModKey, v: number): number {
  const pctStats: ModKey[] = ['crit', 'atkPct', 'defPct', 'hpPct', 'aspd', 'lifesteal', 'skillPct', 'cdPct']
  return pctStats.includes(stat) ? Math.round(v * 1000) / 1000 : Math.round(v)
}

function rollAffixes(rng: () => number, quality: Quality, floor: number): Affix[] {
  const count = QUALITY_AFFIX_COUNT[quality]
  if (count === 0) return []
  const pool = [...AFFIXES]
  const out: Affix[] = []
  const scale = (1 + (floor - 1) * 0.15) * (1 + quality * 0.1)
  for (let i = 0; i < count && pool.length > 0; i++) {
    const idx = randInt(rng, 0, pool.length - 1)
    const def: AffixDef = pool.splice(idx, 1)[0]
    const raw = def.min + rng() * (def.max - def.min)
    out.push({ id: def.id, name: def.name, stat: def.stat, value: roundAffixValue(def.stat, raw * scale) })
  }
  return out
}

/** 机缘词缀：品质词缀之外按概率额外降临 1~2 条（数值更丰厚，UI 标金） */
function rollLuckyAffixes(rng: () => number, quality: Quality, floor: number, luck: number, used: Affix[]): Affix[] {
  const chance = 0.3 + luck * 0.004
  const out: Affix[] = []
  const pool = AFFIXES.filter((a) => !used.some((u) => u.id === a.id))
  const scale = (1 + (floor - 1) * 0.15) * (1 + quality * 0.1) * 1.25
  const count = rng() < chance ? (quality >= 2 && rng() < chance * 0.5 ? 2 : 1) : 0
  for (let i = 0; i < count && pool.length > 0; i++) {
    const idx = randInt(rng, 0, pool.length - 1)
    const def: AffixDef = pool.splice(idx, 1)[0]
    const raw = def.min + rng() * (def.max - def.min)
    out.push({ id: def.id, name: def.name, stat: def.stat, value: roundAffixValue(def.stat, raw * scale), lucky: true })
  }
  return out
}

function buildItem(base: ItemBase, quality: Quality, floor: number, rng: () => number, luck = 0): EquipItem {
  const mult = QUALITY_MULT[quality] * (1 + (floor - 1) * 0.18)
  const affixes = rollAffixes(rng, quality, floor)
  affixes.push(...rollLuckyAffixes(rng, quality, floor, luck, affixes))
  return normalizeItem({
    uid: `${Date.now().toString(36)}-${itemSeq++}`,
    baseId: base.id,
    name: base.name,
    slot: base.slot,
    quality,
    setId: base.set,
    atk: Math.round((base.atk ?? 0) * mult),
    def: Math.round((base.def ?? 0) * mult),
    speed: Math.round((base.speed ?? 0) * mult),
    crit: (base.crit ?? 0) + quality * 0.01,
    maxHp: Math.round((base.maxHp ?? 0) * mult),
    affixes,
    price: Math.round(8 * mult * (quality + 1)),
  })
}

/** GM 工具：按基座 id 直接打造一件指定品质装备 */
export function makeItem(baseId: string, quality: Quality, floor: number): EquipItem | null {
  const base = ITEM_BASES.find((b) => b.id === baseId)
  if (!base) return null
  return buildItem(base, quality, Math.max(1, floor), Math.random)
}

/** 名器化：程序命名 + 基座数值提升 15%~30%（灵品以上才有缘得遇） */
function makeNamed(it: EquipItem, rng: () => number): EquipItem {
  it.name = genNamedName(rng, it.slot)
  it.named = true
  const boost = 1.15 + rng() * 0.15
  it.atk = Math.round(it.atk * boost)
  it.def = Math.round(it.def * boost)
  it.speed = Math.round(it.speed * boost)
  it.maxHp = Math.round(it.maxHp * boost)
  it.price = Math.round(it.price * 2)
  return it
}

/** 普通掉落装备（不含套装件） */
export function rollEquip(rng: () => number, floor: number, luckBonus = 0): EquipItem {
  const pool = ITEM_BASES.filter((b) => !b.set)
  const poolSize = Math.min(pool.length, 6 + floor * 2)
  const base = pool[randInt(rng, 0, poolSize - 1)]
  const entries: Array<[Quality, number]> = ([0, 1, 2, 3] as Quality[]).map((q) => [
    q,
    QUALITY_WEIGHT[q] + (q >= 2 ? luckBonus : 0),
  ])
  const quality = pickWeighted(rng, entries)
  let item = buildItem(base, quality, floor, rng, luckBonus)
  // 名器机缘：灵品以上小概率获得程序命名的名器（基座数值提升）
  if (item.quality >= 1 && rng() < 0.08 + luckBonus * 0.002) item = makeNamed(item, rng)
  return item
}

/** 套装部件（Boss 掉落），quality 灵品~玄品；luck 提高高品质权重 */
export function rollSetPiece(rng: () => number, setId: string, floor: number, luck = 0): EquipItem | null {
  const set = SETS[setId]
  if (!set) return null
  const baseId = set.pieces[randInt(rng, 0, set.pieces.length - 1)]
  const base = ITEM_BASES.find((b) => b.id === baseId)
  if (!base) return null
  const quality = pickWeighted(rng, [[1, Math.max(10, 45 - luck)], [2, 40 + Math.floor(luck / 2)], [3, 15 + luck]] as Array<[Quality, number]>)
  return buildItem(base, quality, floor, rng, luck)
}

export interface DropRoll {
  gold: number
  potion: boolean
  equip: EquipItem | null
}

export function rollDrop(rng: () => number, goldRange: [number, number], floor: number, isBoss: boolean, luck = 0, goldMult = 1): DropRoll {
  const gold = Math.round(randInt(rng, goldRange[0], goldRange[1]) * (isBoss ? 2 : 1) * goldMult)
  const potion = rng() < (isBoss ? 0.9 : 0.14)
  let equip: EquipItem | null = null
  const equipChance = isBoss ? 1 : 0.13 + luck * 0.002  // 难度加成微幅提升装备掉率
  if (rng() < equipChance) equip = rollEquip(rng, floor, (isBoss ? 20 : 0) + luck)
  return { gold, potion, equip }
}

function addMod(mods: Mods, key: ModKey, v: number) { mods[key] += v }

/** 汇总所有来源的修饰 */
export function aggregateMods(s: GameState): Mods {
  const mods = emptyMods()

  // 装备基座数值（含强化）+ 词缀
  for (const key of Object.keys(s.equipment) as SlotKey[]) {
    const it = s.equipment[key]
    if (!it) continue
    const mult = enhMult(it)
    addMod(mods, 'atk', Math.round(it.atk * mult))
    addMod(mods, 'def', Math.round(it.def * mult))
    addMod(mods, 'speed', Math.round(it.speed * mult))
    addMod(mods, 'crit', it.crit + 0.005 * (it.enh ?? 0))
    addMod(mods, 'maxHp', Math.round(it.maxHp * mult))
    for (const af of it.affixes) addMod(mods, af.stat, af.value)
  }

  // 套装阶梯效果
  for (const setId of equippedSetIds(s)) {
    const count = setPieceCount(s, setId)
    for (const b of SETS[setId].bonuses) {
      if (count >= b.count && b.mods) {
        for (const [k, v] of Object.entries(b.mods)) addMod(mods, k as ModKey, v as number)
      }
    }
  }

  // 心法（按功法等级缩放）
  for (const id of s.skillSlots.minds) {
    if (!id) continue
    const m = mindModsAt(id, s.skills[id] ?? 1)
    for (const [k, v] of Object.entries(m)) addMod(mods, k as ModKey, v as number)
  }

  // 天赋
  const tm = talentMods(s)
  for (const [k, v] of Object.entries(tm)) addMod(mods, k as ModKey, v as number)

  return mods
}

export function calcStats(s: GameState): Stats {
  const c = CLASSES[s.classId]
  const lv = s.level - 1
  const m = aggregateMods(s)
  const realmMult = 1 + REALM_STAT_BONUS * s.realm
  return {
    maxHp: Math.round((c.base.hp + c.growth.hp * lv + m.maxHp) * (1 + m.hpPct) * realmMult),
    atk: Math.round((c.base.atk + c.growth.atk * lv + m.atk) * (1 + m.atkPct) * realmMult),
    def: Math.round((c.base.def + c.growth.def * lv + m.def) * (1 + m.defPct) * realmMult),
    speed: c.base.speed + m.speed,
    crit: c.base.crit + m.crit,
    aspd: Math.min(0.4, m.aspd),
    lifesteal: Math.min(0.35, m.lifesteal),
    skillPct: m.skillPct,
    cdPct: Math.min(0.5, m.cdPct),
  }
}

export function currentHp(s: GameState): number {
  const max = calcStats(s).maxHp
  return s.hp < 0 || s.hp > max ? max : s.hp
}

// ---------- 套装 ----------
export function equippedSetIds(s: GameState): string[] {
  const ids = new Set<string>()
  for (const key of Object.keys(s.equipment) as SlotKey[]) {
    const sid = s.equipment[key]?.setId
    if (sid) ids.add(sid)
  }
  return [...ids]
}

export function setPieceCount(s: GameState, setId: string): number {
  let n = 0
  for (const key of Object.keys(s.equipment) as SlotKey[]) {
    if (s.equipment[key]?.setId === setId) n++
  }
  return n
}

// ---------- 攻击模式解析：套装 > 主修招式（需职业匹配） > 职业默认 ----------
function classDefaultMode(s: GameState): AttackMode {
  const def = SKILLS.get(CLASSES[s.classId].defaultStyle)
  return def?.attackMode ?? 'melee_arc'
}

/** 当前装配招式是否与职业匹配 */
function styleMatchesClass(s: GameState): boolean {
  const id = s.skillSlots.style
  if (!id) return false
  const def = SKILLS.get(id)
  return !!def && (def.class === 'any' || def.class === s.classId)
}

export function resolveAttackMode(s: GameState): AttackMode {
  for (const setId of equippedSetIds(s)) {
    const count = setPieceCount(s, setId)
    for (const b of SETS[setId].bonuses) {
      if (b.attackMode && count >= b.count) return b.attackMode
    }
  }
  if (styleMatchesClass(s)) {
    const def = SKILLS.get(s.skillSlots.style!)
    if (def?.attackMode) return def.attackMode
  }
  return classDefaultMode(s)
}

/** 套装赋予的神通形态（如凡人套 4 件 → 大庚剑阵）；无则 null */
export function resolveSetArtMode(s: GameState): ArtMode | null {
  for (const setId of equippedSetIds(s)) {
    const count = setPieceCount(s, setId)
    for (const b of SETS[setId].bonuses) {
      if (b.artMode && count >= b.count) return b.artMode
    }
  }
  return null
}

/** 当前招式功法威力（含等级成长）；职业不匹配时按默认招式 1.0 */
export function stylePower(s: GameState): number {
  if (!styleMatchesClass(s)) return 1
  const id = s.skillSlots.style!
  const def = SKILLS.get(id)
  if (!def) return 1
  return (def.power ?? 1) * (1 + 0.12 * ((s.skills[id] ?? 1) - 1))
}

/** 旧版本迁移至 v3 */
function migrate(raw: Record<string, unknown>): GameState {
  const s = newGame()
  s.level = Number(raw.level) || 1
  s.exp = Number(raw.exp) || 0
  s.gold = Number(raw.gold) || 0
  s.potions = Number(raw.potions ?? 3)
  s.insight = Number(raw.insight ?? (s.level - 1) * 3)
  s.hp = typeof raw.hp === 'number' ? raw.hp : -1
  s.classId = String(raw.classId ?? 'sword')
  s.heroName = String(raw.heroName ?? s.heroName)
  s.realm = Number(raw.realm ?? 0) || 0
  s.companion = Boolean(raw.companion)
  const eq = (raw.equipment ?? {}) as Record<string, EquipItem | null>
  s.equipment.weapon = eq.weapon ? normalizeItem(eq.weapon) : null
  s.equipment.head = eq.head ? normalizeItem(eq.head) : null
  s.equipment.armor = eq.armor ? normalizeItem(eq.armor) : null
  s.equipment.boots = eq.boots ? normalizeItem(eq.boots) : null
  s.equipment.charm1 = (eq.charm1 ?? eq.charm) ? normalizeItem((eq.charm1 ?? eq.charm)!) : null
  s.equipment.charm2 = eq.charm2 ? normalizeItem(eq.charm2) : null
  if (Array.isArray(raw.inventory)) s.inventory = (raw.inventory as EquipItem[]).map(normalizeItem)
  const sk = (raw.skills ?? {}) as Record<string, number>
  if (Object.keys(sk).length > 0) s.skills = { ...sk }
  const slots = raw.skillSlots as SkillSlots | undefined
  if (slots) s.skillSlots = { style: slots.style ?? null, minds: slots.minds ?? [null, null], art: slots.art ?? null }
  s.talents = { ...((raw.talents ?? {}) as Record<string, number>) }
  s.progress = {
    qmCleared: Boolean((raw.progress as { qmCleared?: boolean } | undefined)?.qmCleared),
    ymCleared: Boolean((raw.progress as { ymCleared?: boolean } | undefined)?.ymCleared),
  }
  s.favor = { ...((raw.favor ?? {}) as Record<string, number>) }
  s.quests = { ...((raw.quests ?? {}) as Record<string, 'active' | 'done'>) }
  s.totalKills = Number(raw.totalKills ?? 0) || 0
  if (Array.isArray(raw.presets)) {
    s.presets = (raw.presets as Array<BuildPreset | null>).slice(0, PRESET_COUNT)
    while (s.presets.length < PRESET_COUNT) s.presets.push(null)
  }
  s.activePreset = typeof raw.activePreset === 'number' ? raw.activePreset : -1
  s.syCheckpoint = Number(raw.syCheckpoint ?? 7) || 7
  s.syMaxDepth = Number(raw.syMaxDepth ?? 0) || 0
  return s
}

export function loadGame(): GameState {
  try {
    const raw = localStorage.getItem(SAVE_KEY)
    if (!raw) return newGame()
    const parsed = JSON.parse(raw) as Record<string, unknown>
    if (parsed.version === SAVE_VERSION) {
      const s = parsed as unknown as GameState
      for (const k of Object.keys(s.equipment) as SlotKey[]) {
        if (s.equipment[k]) normalizeItem(s.equipment[k]!)
      }
      s.inventory = (s.inventory ?? []).map(normalizeItem)
      s.progress = { qmCleared: s.progress?.qmCleared ?? false, ymCleared: s.progress?.ymCleared ?? false }
      s.materials = s.materials ?? { tupo: 0 }
      s.favor = s.favor ?? {}
      s.quests = s.quests ?? {}
      s.totalKills = s.totalKills ?? 0
      if (!Array.isArray(s.presets)) s.presets = [null, null, null, null, null]
      while (s.presets.length < PRESET_COUNT) s.presets.push(null)
      s.activePreset = s.activePreset ?? -1
      s.syCheckpoint = s.syCheckpoint ?? 7
      s.syMaxDepth = s.syMaxDepth ?? 0
      s.dynamicSets = s.dynamicSets ?? []
      s.dynamicSkills = s.dynamicSkills ?? []
      s.codex = s.codex ?? { items: [], skills: [], sets: [] }
      s.codex.items = s.codex.items ?? []
      s.codex.skills = s.codex.skills ?? []
      s.codex.sets = s.codex.sets ?? []
      registerDynamicContent(s.dynamicSets, s.dynamicSkills)
      return s
    }
    if (typeof parsed.version === 'number' && parsed.version < SAVE_VERSION) return migrate(parsed)
    return newGame()
  } catch {
    return newGame()
  }
}

/** 加经验，返回升了几级（悟道点由调用方发放） */
export function gainExp(s: GameState, amount: number): number {
  s.exp += amount
  let ups = 0
  while (s.exp >= expNeed(s.level)) {
    s.exp -= expNeed(s.level)
    s.level++
    ups++
  }
  return ups
}

// ---------- 机缘套装（程序生成） ----------

export interface DynamicSetRecord { set: SetDef; bases: ItemBase[] }

const SET_PIECE_SLOTS: ItemSlot[] = ['weapon', 'head', 'armor', 'boots']
const SET_PIECE_SUFFIX: Record<ItemSlot, string[]> = {
  weapon: ['剑', '刃', '环', '印'],
  head: ['冠', '簪', '巾'],
  armor: ['袍', '甲', '衣'],
  boots: ['靴', '履', '翅'],
  charm: ['佩'],
}

/** 套装阶梯修饰候选（min, max） */
const SET_MOD_POOL: Array<[ModKey, number, number]> = [
  ['atkPct', 0.10, 0.18], ['defPct', 0.12, 0.20], ['hpPct', 0.10, 0.18],
  ['crit', 0.03, 0.06], ['aspd', 0.05, 0.10], ['skillPct', 0.10, 0.18],
  ['lifesteal', 0.03, 0.06], ['speed', 6, 12], ['cdPct', 0.06, 0.12],
]

function rollSetMods(rng: () => number, count: number, scale: number): Partial<Mods> {
  const pool = [...SET_MOD_POOL]
  const out: Partial<Mods> = {}
  for (let i = 0; i < count && pool.length > 0; i++) {
    const [key, lo, hi] = pool.splice(ri(rng, 0, pool.length - 1), 1)[0]
    const v = (lo + rng() * (hi - lo)) * scale
    ;(out as Record<string, number>)[key] = key === 'speed' ? Math.round(v) : round3(v)
  }
  return out
}

function modsDesc(mods: Partial<Mods>): string {
  return (Object.entries(mods) as Array<[ModKey, number]>).map(([k, v]) => MOD_LABELS[k](v)).join('，')
}

/** 生成一套机缘套装（4 件，纯数值加成，不改变招式形态） */
export function genDynamicSet(rng: () => number, seq: number, existing: DynamicSetRecord[]): DynamicSetRecord {
  const used = new Set(existing.map((r) => r.set.name.replace(/套$/, '')))
  const free = SET_THEMES.filter((t) => !used.has(t))
  const theme = free.length > 0 ? pick(rng, free) : pick(rng, SET_THEMES) + seq
  const setId = `set_dy${seq}`
  const tier = 1 + Math.min(0.5, seq * 0.06) // 越往后生成的套装基座越强
  const bases: ItemBase[] = SET_PIECE_SLOTS.map((slot) => {
    const id = `${slot[0]}_dy${seq}`
    const name = theme + pick(rng, SET_PIECE_SUFFIX[slot])
    const base: ItemBase = { id, name, slot, set: setId }
    if (slot === 'weapon') { base.atk = ri(rng, 19, 25); if (rng() < 0.6) base.crit = round3(0.03 + rng() * 0.03) }
    if (slot === 'head') { base.def = ri(rng, 5, 7); base.maxHp = ri(rng, 22, 34); if (rng() < 0.5) base.crit = 0.02 }
    if (slot === 'armor') { base.def = ri(rng, 11, 14); base.maxHp = ri(rng, 20, 32); if (rng() < 0.4) base.speed = ri(rng, 3, 5) }
    if (slot === 'boots') { base.speed = ri(rng, 12, 17); base.def = ri(rng, 3, 5) }
    if (base.atk) base.atk = Math.round(base.atk * tier)
    if (base.def) base.def = Math.round(base.def * tier)
    if (base.maxHp) base.maxHp = Math.round(base.maxHp * tier)
    if (base.speed) base.speed = Math.round(base.speed * tier)
    return base
  })
  const m2 = rollSetMods(rng, ri(rng, 1, 2), 1)
  const m4 = rollSetMods(rng, 2, 1.8)
  const set: SetDef = {
    id: setId,
    name: `${theme}套`,
    pieces: bases.map((b) => b.id),
    bonuses: [
      { count: 2, mods: m2, desc: `2件：${modsDesc(m2)}` },
      { count: 4, mods: m4, desc: `4件：${modsDesc(m4)}` },
    ],
  }
  return { set, bases }
}

const MIND_MOD_POOL: Array<[ModKey, number, number]> = [
  ['atkPct', 0.08, 0.16], ['defPct', 0.10, 0.20], ['hpPct', 0.08, 0.16],
  ['crit', 0.03, 0.07], ['aspd', 0.04, 0.09], ['skillPct', 0.08, 0.16],
  ['cdPct', 0.06, 0.14], ['lifesteal', 0.02, 0.05], ['speed', 4, 10],
]

/** 生成一部无名古卷心法（纯数值，通用心法） */
export function genDynamicSkill(rng: () => number, seq: number): SkillDef {
  const name = pick(rng, SKILL_PREFIX) + pick(rng, SKILL_SUFFIX)
  const pool = [...MIND_MOD_POOL]
  const mods: Partial<Mods> = {}
  const n = ri(rng, 1, 2)
  for (let i = 0; i < n && pool.length > 0; i++) {
    const [key, lo, hi] = pool.splice(ri(rng, 0, pool.length - 1), 1)[0]
    const v = lo + rng() * (hi - lo)
    ;(mods as Record<string, number>)[key] = key === 'speed' ? Math.round(v) : round3(v)
  }
  return {
    id: `sk_dy${seq}`,
    name: SKILLS.has(`sk_dy${seq}`) ? `${name}·${seq}` : name,
    type: 'mind',
    class: 'any',
    mods,
    desc: `无名古卷所录心法：${modsDesc(mods)}。`,
  }
}

export function registerDynamicContent(sets: DynamicSetRecord[] | undefined, skills: SkillDef[] | undefined) {
  for (const rec of sets ?? []) {
    if (!rec?.set?.id) continue
    if (!SETS[rec.set.id]) SETS[rec.set.id] = rec.set
    for (const base of rec.bases ?? []) {
      if (!base?.id || ITEM_BASE_MAP.has(base.id)) continue
      base.set = rec.set.id
      ITEM_BASES.push(base)
      ITEM_BASE_MAP.set(base.id, base)
    }
  }
  for (const sk of skills ?? []) {
    if (sk?.id && !SKILLS.has(sk.id)) SKILLS.set(sk.id, sk)
  }
}

export const SKILL_MAX_LEVEL = 5

export function skillDef(id: string): SkillDef | undefined { return SKILLS.get(id) }

export function skillLevel(s: GameState, id: string): number { return s.skills[id] ?? 0 }

/** 升级消耗悟道点：当前等级 × 2 */
export function upgradeCost(level: number): number { return level * 2 }

export function canUpgrade(s: GameState, id: string): boolean {
  const lv = skillLevel(s, id)
  return lv >= 1 && lv < SKILL_MAX_LEVEL && s.insight >= upgradeCost(lv)
}

export function upgradeSkill(s: GameState, id: string): boolean {
  if (!canUpgrade(s, id)) return false
  s.insight -= upgradeCost(s.skills[id])
  s.skills[id]++
  return true
}

/** 招式/神通威力（含等级成长 +12%/级） */
export function skillPowerAt(s: GameState, id: string): number {
  const def = SKILLS.get(id)
  if (!def) return 1
  return (def.power ?? 1) * (1 + 0.12 * (skillLevel(s, id) - 1))
}

/** 心法修饰（含等级成长 +20%/级） */
export function mindModsAt(id: string, level: number): Partial<Mods> {
  const def = SKILLS.get(id)
  if (!def?.mods) return {}
  const scale = 1 + 0.2 * (level - 1)
  const out: Partial<Mods> = {}
  for (const [k, v] of Object.entries(def.mods)) {
    ;(out as Record<string, number>)[k] = Math.round((v as number) * scale * 1000) / 1000
  }
  return out
}

/** 神通冷却（含天赋冷却缩减） */
export function artCooldown(s: GameState, id: string, cdPct: number): number {
  const def = SKILLS.get(id)
  if (!def?.cd) return 5000
  return def.cd * (1 - cdPct)
}

export type EquipSkillResult = 'ok' | 'full' | 'bad_type' | 'not_learned'

/** 装配功法到对应槽位（职业不匹配的招式/神通不可装配） */
export function equipSkill(s: GameState, id: string): EquipSkillResult {
  const def = SKILLS.get(id)
  if (!def || skillLevel(s, id) < 1) return 'not_learned'
  if (def.type !== 'mind' && def.class !== 'any' && def.class !== s.classId) return 'bad_type'
  if (def.type === 'style') { s.skillSlots.style = id; return 'ok' }
  if (def.type === 'art') { s.skillSlots.art = id; return 'ok' }
  // 心法：先填空位
  if (s.skillSlots.minds.includes(id)) return 'ok'
  const empty = s.skillSlots.minds.indexOf(null)
  if (empty === -1) return 'full'
  s.skillSlots.minds[empty] = id
  return 'ok'
}

export function unequipSkill(s: GameState, id: string): void {
  if (s.skillSlots.style === id) s.skillSlots.style = null
  if (s.skillSlots.art === id) s.skillSlots.art = null
  const i = s.skillSlots.minds.indexOf(id)
  if (i >= 0) s.skillSlots.minds[i] = null
}

export function isSkillEquipped(s: GameState, id: string): boolean {
  return s.skillSlots.style === id || s.skillSlots.art === id || s.skillSlots.minds.includes(id)
}

/** 未习得且当前职业可用的功法 id 列表（用于掉落/商店） */
export function unlearnedSkills(s: GameState): string[] {
  return [...SKILLS.values()]
    .filter((d) => !(d.id in s.skills) && (d.class === 'any' || d.class === s.classId))
    .map((d) => d.id)
}

export function learnSkill(s: GameState, id: string): void {
  if (!(id in s.skills)) s.skills[id] = 1
  recordCodexSkill(s, id)
}

import { TALENT_NODES, emptyMods, type ModKey, type Mods } from './data'
import type { GameState } from './state'

export const RESPEC_COST = 100

export function spentPoints(s: GameState): number {
  return Object.values(s.talents).reduce((a, b) => a + b, 0)
}

/** 可用天赋点：每级 1 点 */
export function availablePoints(s: GameState): number {
  return Math.max(0, s.level - 1 - spentPoints(s))
}

export function addPoint(s: GameState, nodeId: string): boolean {
  const node = TALENT_NODES.get(nodeId)
  if (!node) return false
  const cur = s.talents[nodeId] ?? 0
  if (cur >= node.max || availablePoints(s) <= 0) return false
  s.talents[nodeId] = cur + 1
  return true
}

export function respec(s: GameState): boolean {
  if (s.gold < RESPEC_COST || spentPoints(s) === 0) return false
  s.gold -= RESPEC_COST
  s.talents = {}
  return true
}

/** 天赋总修饰 */
export function talentMods(s: GameState): Mods {
  const mods = emptyMods()
  for (const [id, pts] of Object.entries(s.talents)) {
    const node = TALENT_NODES.get(id)
    if (!node || pts <= 0) continue
    for (const [k, v] of Object.entries(node.per)) {
      mods[k as ModKey] += (v as number) * pts
    }
  }
  return mods
}

export type Facing = 'down' | 'left' | 'right' | 'up'

export class Player {
  sprite: Phaser.Physics.Arcade.Sprite
  dir = new Phaser.Math.Vector2(0, 1)
  facing: Facing = 'down'
  attackCdUntil = 0
  comboStage = 0
  comboResetAt = 0
  hurtUntil = 0
  private animLockUntil = 0
  private classId: string
  private animated: boolean

  constructor(scene: Phaser.Scene, x: number, y: number, classId = 'sword') {
    this.classId = classId
    this.animated = scene.anims.exists(`${classId}_down_idle`)
    if (this.animated) {
      this.sprite = scene.physics.add.sprite(x, y, `${classId}_atlas`, `${classId}_down_idle_00`)
      this.sprite.setScale(0.55) // 64px 帧 → 约 35px 显示高度
      const body = this.sprite.body as Phaser.Physics.Arcade.Body
      body.setSize(24, 14)
      body.setOffset(20, 48)
      this.playAnim('idle')
    } else {
      this.sprite = scene.physics.add.sprite(x, y, 'player')
      const body = this.sprite.body as Phaser.Physics.Arcade.Body
      body.setSize(10, 8)
      body.setOffset(3, 8)
    }
    this.sprite.setDepth(10)
  }

  /** 动画使用的朝向（按职业配置修正素材方向标反的情况） */
  private animFacing(): Facing {
    return (CLASSES[this.classId]?.dirRemap?.[this.facing] ?? this.facing) as Facing
  }

  private playAnim(action: 'idle' | 'walk' | 'attack' | 'hit' | 'cast') {
    if (!this.animated) return
    const key = `${this.classId}_${this.animFacing()}_${action}`
    if (this.sprite.scene.anims.exists(key)) this.sprite.play(key, true)
  }

  move(input: InputPort, stats: Stats, time: number) {
    const { x, y } = input.move
    const moving = x !== 0 || y !== 0
    if (moving) {
      this.dir.set(x, y).normalize()
      this.sprite.setVelocity(x * stats.speed, y * stats.speed)
      if (!this.animated) {
        if (x !== 0) this.sprite.setFlipX(x < 0)
      } else {
        this.facing = Math.abs(x) >= Math.abs(y) ? (x < 0 ? 'left' : 'right') : (y < 0 ? 'up' : 'down')
      }
    } else {
      this.sprite.setVelocity(0, 0)
    }
    if (time < this.animLockUntil) return
    this.playAnim(moving ? 'walk' : 'idle')
  }

  halt() { this.sprite.setVelocity(0, 0) }

  /** 返回连击段数（1~3），未准备好返回 0；aspd 为攻速加成（0~0.4） */
  tryAttack(time: number, aspd = 0): number {
    if (time < this.attackCdUntil) return 0
    this.comboStage = time < this.comboResetAt ? (this.comboStage % 3) + 1 : 1
    this.comboResetAt = time + 950
    this.attackCdUntil = time + 370 * (1 - aspd)
    this.animLockUntil = time + 360
    this.playAnim('attack')
    return this.comboStage
  }

  playHit(time: number) {
    this.animLockUntil = Math.max(this.animLockUntil, time + 280)
    this.playAnim('hit')
  }

  /** 神通施法动画（剑修有 cast 序列；无 cast 动画的职业退化为 attack） */
  playCast(time: number) {
    this.animLockUntil = Math.max(this.animLockUntil, time + 500)
    const key = `${this.classId}_${this.animFacing()}_cast`
    this.playAnim(this.sprite.scene.anims.exists(key) ? 'cast' : 'attack')
  }
}

export type EnemyState = 'idle' | 'chase' | 'windup' | 'dash' | 'dead'

export class Enemy {
  sprite: Phaser.Physics.Arcade.Sprite
  cfg: MonsterDef
  hp: number
  maxHp: number
  elite = false   // 精英：更高属性，必掉装备
  atk: number
  def: number
  speed: number
  exp: number
  gold: [number, number]
  isBoss: boolean
  homeX: number             // 出生点（必为可行走格）：越界时的回拉锚点
  homeY: number
  state: EnemyState = 'idle'
  slowUntil = 0
  private cdUntil = 0
  private windupUntil = 0
  private wanderUntil = 0
  private wanderDir = new Phaser.Math.Vector2(0, 0)
  private dashUntil = 0
  private nextDashAt = 0
  private dashHit = false
  private lastX = 0
  private lastY = 0
  private stuckCheckAt = 0
  private unstuckUntil = 0

  constructor(scene: Phaser.Scene, cfg: MonsterDef, x: number, y: number, scale: number) {
    this.cfg = cfg
    this.homeX = x
    this.homeY = y
    this.maxHp = Math.round(cfg.hp * scale)
    this.hp = this.maxHp
    this.atk = Math.round(cfg.atk * (0.8 + scale * 0.2))
    this.def = cfg.def + Math.floor((scale - 1) * 2)
    this.speed = cfg.speed
    this.exp = Math.round(cfg.exp * scale)
    this.gold = cfg.gold
    this.isBoss = cfg.ai === 'boss'
    this.sprite = scene.physics.add.sprite(x, y, cfg.tex)
    this.sprite.setDepth(9)
    this.sprite.setScale(cfg.scale ?? 1)
    this.sprite.setCollideWorldBounds(true)   // 世界边界硬阻挡，防止被击退/冲撞挤出地图
    const body = this.sprite.body as Phaser.Physics.Arcade.Body
    // 碰撞体按贴图帧尺寸自适应：水平居中、底部对齐（16px 帧 = offset(2,6)，32px Boss 帧 = offset(10,22)）
    const fw = this.sprite.frame.width
    const fh = this.sprite.frame.height
    const bw = Math.min(12, fw)
    const bh = Math.min(10, fh)
    body.setSize(bw, bh)
    body.setOffset((fw - bw) / 2, fh - bh)
  }

  get x() { return this.sprite.x }
  get y() { return this.sprite.y }
  get alive() { return this.state !== 'dead' }

  /** 减速（冰霜类效果） */
  applySlow(ms: number, time: number) {
    if (!this.alive) return
    this.slowUntil = Math.max(this.slowUntil, time + ms)
    this.sprite.setTint(0x9fd8ff)
    this.sprite.scene.time.delayedCall(ms, () => {
      if (this.alive && this.state !== 'windup' && time + ms >= this.slowUntil) this.sprite.clearTint()
    })
  }

  takeDamage(dmg: number, fromX: number, fromY: number): boolean {
    if (!this.alive) return false
    this.hp -= dmg
    this.sprite.setTintFill(0xffffff)
    this.sprite.scene.time.delayedCall(70, () => { if (this.alive) this.sprite.clearTint() })
    // 轻微击退
    const kb = new Phaser.Math.Vector2(this.x - fromX, this.y - fromY).normalize().scale(this.isBoss ? 20 : 70)
    this.sprite.setVelocity(kb.x, kb.y)
    if (this.hp <= 0) { this.die(); return true }
    return false
  }

  private die() {
    this.state = 'dead'
    this.sprite.setVelocity(0, 0)
    this.sprite.setTint(0x555555)
    this.sprite.scene.tweens.add({
      targets: this.sprite, alpha: 0, y: this.sprite.y + 6, duration: 380,
      onComplete: () => this.sprite.destroy(),
    })
  }

  update(ctx: CombatCtx) {
    if (!this.alive) return
    const { time, px, py } = ctx
    const dist = Phaser.Math.Distance.Between(this.x, this.y, px, py)
    const toPlayer = new Phaser.Math.Vector2(px - this.x, py - this.y)
    const aggro = dist < 210 || this.hp < this.maxHp

    // 卡墙检测：追击中 350ms 位移不足 3px，则随机侧移 450ms 脱困
    if (time >= this.stuckCheckAt) {
      const moved = Phaser.Math.Distance.Between(this.x, this.y, this.lastX, this.lastY)
      if (this.state === 'chase' && this.speed > 0 && aggro && moved < 3) {
        this.unstuckUntil = time + 450
        this.wanderDir.set(Math.random() * 2 - 1, Math.random() * 2 - 1).normalize()
      }
      this.lastX = this.x
      this.lastY = this.y
      this.stuckCheckAt = time + 350
    }

    if (this.x < px - 2) this.sprite.setFlipX(false)
    else if (this.x > px + 2) this.sprite.setFlipX(true)

    // 出招中
    if (this.state === 'windup') {
      this.sprite.setVelocity(0, 0)
      if (time >= this.windupUntil) {
        this.state = 'chase'
        this.sprite.clearTint()
        this.cdUntil = time + (this.cfg.cd ?? 1200)
        if (this.cfg.ai === 'ranged') {
          ctx.spawnProj(this.x, this.y - 4, toPlayer.angle(), this.atk)
        } else {
          const reach = this.isBoss ? 48 : 32
          if (dist < reach) ctx.damagePlayer(this.atk, this.x, this.y)
        }
      }
      return
    }

    // Boss 冲撞
    if (this.state === 'dash') {
      if (!this.dashHit && dist < 40) {
        this.dashHit = true
        ctx.damagePlayer(Math.round(this.atk * 1.4), this.x, this.y)
      }
      if (time >= this.dashUntil) { this.state = 'chase'; this.sprite.clearTint() }
      return
    }

    if (!aggro) {
      // 游荡
      const sp = this.speed * (time < this.slowUntil ? 0.5 : 1)
      if (time > this.wanderUntil) {
        this.wanderUntil = time + 1200 + Math.random() * 1800
        if (Math.random() < 0.45) this.wanderDir.set(0, 0)
        else this.wanderDir.set(Math.random() * 2 - 1, Math.random() * 2 - 1).normalize()
      }
      this.sprite.setVelocity(this.wanderDir.x * sp * 0.35, this.wanderDir.y * sp * 0.35)
      return
    }

    // Boss 冲撞触发
    if (this.isBoss && time > this.nextDashAt && dist > 70 && dist < 240) {
      this.nextDashAt = time + 6500
      this.dashHit = false
      this.state = 'dash'
      this.dashUntil = time + 520
      this.sprite.setTint(0xff6666)
      const dash = toPlayer.clone().normalize().scale(280)
      this.sprite.setVelocity(dash.x, dash.y)
      return
    }

    const cdReady = time >= this.cdUntil
    const sp = this.speed * (time < this.slowUntil ? 0.5 : 1)

    // 脱困移动中：径直侧移，绕开障碍
    if (time < this.unstuckUntil) {
      this.sprite.setVelocity(this.wanderDir.x * sp, this.wanderDir.y * sp)
      return
    }

    if (this.cfg.ai === 'ranged') {
      const range = this.cfg.range ?? 160
      if (dist < 70) {
        const away = toPlayer.clone().normalize().scale(-sp)
        this.sprite.setVelocity(away.x, away.y)
      } else if (dist <= range && cdReady) {
        this.state = 'windup'
        this.windupUntil = time + 420
        this.sprite.setTint(0xffe08a)
      } else if (dist > range) {
        const t = toPlayer.clone().normalize().scale(sp)
        this.sprite.setVelocity(t.x, t.y)
      } else {
        this.sprite.setVelocity(0, 0)
      }
      return
    }

    // 近战（含 Boss 常态）
    const reach = this.isBoss ? 46 : 28
    if (dist <= reach && cdReady) {
      this.state = 'windup'
      this.windupUntil = time + (this.isBoss ? 420 : 320)
      this.sprite.setTint(0xffe08a)
    } else if (dist > reach - 4) {
      const t = toPlayer.clone().normalize().scale(sp)
      this.sprite.setVelocity(t.x, t.y)
    } else {
      this.sprite.setVelocity(0, 0)
    }
  }
}
